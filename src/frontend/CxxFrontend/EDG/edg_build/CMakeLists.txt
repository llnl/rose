# ROSE's configuration of the EDG C/C++ front end, in the format of EDG's own build system.
#
# EDG's CMake project (../edg_compiler) configures the front end from "macro configuration
# sets": directories of *.cmakedef files (NAME=VALUE lines plus "import <...>") that its
# dev_tools/bin/edg-cmakedef resolves into the generated cmake_defines.h, which EDG's
# src/defines.h includes when USE_CMAKE_DEFINES is defined.  This directory produces the set
# "rose" in the build tree:
#
#   <build>/macro-conf/rose/base.cmakedef  copy of macro-conf/rose/base.cmakedef: the
#                                          host-independent part of the configuration
#   <build>/macro-conf/rose/host.cmakedef  generated below: the parts that depend on the
#                                          host, on the backend compiler ROSE was configured
#                                          for and on ROSE's options, followed by the target
#                                          description printed by EDG's dettarg utility
#   <build>/macro-conf/rose/rose.cmakedef  copy: per-target definitions (none)
#
# ../CMakeLists.txt points EDG's build at this set (EDG_MACRO_CONF_ROOT, EDG_MACRO_CONF) and
# creates the library with edg_add_frontend_library(edg_frontend rose).
#
# base + host replace the former "defines.h.rose + dettarg.out -> cmake_defines.h"
# concatenation one to one (a later cmakedef line overrides an earlier one, as a later
# #define did); defines.h.rose is kept in this directory only for the deprecated autotools
# build (Makefile.am).  The conditions below are the #if's of defines.h.rose; the comments
# name the original preprocessor tests.

set(EDG_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../edg_compiler")

set(conf_dir "${CMAKE_CURRENT_BINARY_DIR}/macro-conf/rose")
file(MAKE_DIRECTORY "${conf_dir}")
configure_file(macro-conf/rose/base.cmakedef "${conf_dir}/base.cmakedef" COPYONLY)
configure_file(macro-conf/rose/rose.cmakedef "${conf_dir}/rose.cmakedef" COPYONLY)

set(host_defs "# Generated by ${CMAKE_CURRENT_LIST_FILE} -- do not edit.\n")
macro(rose_edg_define name value)
  string(APPEND host_defs "${name}=${value}\n")
endmacro()
macro(rose_edg_undefine name)
  string(APPEND host_defs "delete_if_set ${name}\n")
endmacro()
macro(rose_edg_comment text)
  string(APPEND host_defs "\n# ${text}\n")
endmacro()

#------------------------------------------------------------------------------------------
# Host representation of integer values (depends on the backend compiler; __WORDSIZE == 64
# of the original is approximated by an 8-byte pointer)
#------------------------------------------------------------------------------------------
rose_edg_comment("Integer value representation (backend compiler)")
set(int_value_type "unsigned long long")
set(signed_int_value_type "long long")
set(int128_allowed "")
if(MSVC)                                                        # _MSC_VER
  # defaults
elseif(BACKEND_CXX_IS_GNU_COMPILER OR BACKEND_CXX_IS_INTEL_COMPILER)
  if(CMAKE_SIZEOF_VOID_P EQUAL 8)                               # __WORDSIZE == 64
    set(int_value_type "unsigned __int128")
    set(signed_int_value_type "__int128")
    set(int128_allowed 1)
  endif()
elseif(BACKEND_CXX_IS_CLANG_COMPILER)
  if(NOT rose_edg_arch STREQUAL arm64)                          # TARG_SUPPORTS_ARM64 != 1
    set(int_value_type "unsigned __int128")
    set(signed_int_value_type "__int128")
    set(int128_allowed 1)
  else()
    set(int128_allowed 0)
  endif()
endif()
rose_edg_define(TYPE_FOR_AN_INTEGER_VALUE "${int_value_type}")
rose_edg_define(TYPE_FOR_A_SIGNED_INTEGER_VALUE "${signed_int_value_type}")
if(NOT int128_allowed STREQUAL "")
  rose_edg_define(INT128_EXTENSIONS_ALLOWED ${int128_allowed})
endif()

  # Approximate __float128 conversions using long double routines instead of relying on the

#------------------------------------------------------------------------------------------
# Host long double representation (EDG 7.0's float_pt_init() needs exactly one of these when
# USE_FLOAT128_FOR_HOST_FP_VALUE is set).  The Apple arm64 block above already chose
# FP_LONG_DOUBLE_IS_BINARY64; otherwise use the compiler's __LDBL_MANT_DIG__ and fall back
# to the architecture.
#------------------------------------------------------------------------------------------
if(NOT (rose_edg_arch STREQUAL arm64 AND APPLE))
  rose_edg_comment("Host long double format (__LDBL_MANT_DIG__ = ${host_ldbl_mant_dig})")
  if(host_ldbl_mant_dig STREQUAL "64")
    rose_edg_define(FP_LONG_DOUBLE_IS_80BIT_EXTENDED 1)
  elseif(host_ldbl_mant_dig STREQUAL "113")
    rose_edg_define(FP_LONG_DOUBLE_IS_BINARY128 1)
  elseif(host_ldbl_mant_dig STREQUAL "53")
    rose_edg_define(FP_LONG_DOUBLE_IS_BINARY64 1)
  elseif(rose_edg_arch STREQUAL arm64 OR host_cpu MATCHES "^riscv")
    rose_edg_define(FP_LONG_DOUBLE_IS_BINARY128 1)
  else()
    rose_edg_define(FP_LONG_DOUBLE_IS_80BIT_EXTENDED 1)
  endif()
endif()

#------------------------------------------------------------------------------------------
# Target description from EDG's dettarg utility, run on the host at configure time (it used
# to run at build time, before the two files were concatenated).  Besides whether the host has
# "long long" (SIZEOF_LONG_LONG from cmake/ConfigureChecks.cmake), ROSE's dettarg patch reads
# ROSE_USE_MICROSOFT_EXTENSIONS and, for HOST_ALIGNMENT_REQUIRED, the backend compiler's
# identity and version; it used to get them from rose_config.h, which does not exist yet at
# this point, so pass exactly what rose_config.h.in.cmake would define.
#------------------------------------------------------------------------------------------
if(SIZEOF_LONG_LONG STREQUAL "")
  set(edg_have_long_long 0)
else()
  set(edg_have_long_long 1)
endif()
set(dettarg_definitions -DHAVE_LONG_LONG=${edg_have_long_long})
if(ROSE_USE_MICROSOFT_EXTENSIONS)
  list(APPEND dettarg_definitions -DROSE_USE_MICROSOFT_EXTENSIONS)
endif()
foreach(var BACKEND_CXX_IS_GNU_COMPILER BACKEND_CXX_IS_CLANG_COMPILER BACKEND_CXX_IS_INTEL_COMPILER)
  if(${var})                                                    # #cmakedefine X @X@
    list(APPEND dettarg_definitions "-D${var}=${${var}}")
  endif()
endforeach()
foreach(var BACKEND_CXX_COMPILER_MAJOR_VERSION_NUMBER BACKEND_CXX_COMPILER_MINOR_VERSION_NUMBER)
  if(DEFINED ${var} AND NOT "${${var}}" STREQUAL "")             # #define X @X@
    list(APPEND dettarg_definitions "-D${var}=${${var}}")
  endif()
endforeach()
# dettarg.c is C++ like the rest of EDG; give it a .cc name so try_run compiles it as such.
configure_file("${EDG_SOURCE_DIR}/util/dettarg.c" "${CMAKE_CURRENT_BINARY_DIR}/dettarg.cc" COPYONLY)
try_run(dettarg_run_result dettarg_compile_result
        "${CMAKE_CURRENT_BINARY_DIR}/dettarg-try"
        "${CMAKE_CURRENT_BINARY_DIR}/dettarg.cc"
        COMPILE_DEFINITIONS ${dettarg_definitions}
        COMPILE_OUTPUT_VARIABLE dettarg_compile_output
        RUN_OUTPUT_VARIABLE dettarg_output)
if(NOT dettarg_compile_result)
  message(FATAL_ERROR "EDG's dettarg utility did not compile:\n${dettarg_compile_output}")
endif()
if(NOT dettarg_run_result EQUAL 0)
  message(FATAL_ERROR "EDG's dettarg utility failed (exit ${dettarg_run_result}):\n${dettarg_output}")
endif()
rose_edg_comment("Target description determined by EDG's dettarg utility")
string(REPLACE ";" "\\;" dettarg_output "${dettarg_output}")
string(REPLACE "\n" ";" dettarg_lines "${dettarg_output}")
foreach(line IN LISTS dettarg_lines)
  if(line MATCHES "^[ \t]*#define[ \t]+([A-Za-z0-9_]+)[ \t]*(.*)$")
    set(name "${CMAKE_MATCH_1}")
    string(STRIP "${CMAKE_MATCH_2}" value)
    rose_edg_define(${name} "${value}")
  endif()
endforeach()
if(SIZEOF_LONG_LONG STREQUAL "")
  rose_edg_define(INTEGER_VALUE_REPR_IS_A_HOST_INTEGER 0)
endif()

                "${conf_dir}/host.cmakedef.tmp" "${conf_dir}/host.cmakedef")
file(REMOVE "${conf_dir}/host.cmakedef.tmp")

               "microsoft extensions ${rose_edg_microsoft})")
