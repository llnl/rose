remove_definitions(-DROSE_DLL_EXPORTS)

# The EDG C/C++ front end comes from the edg_compiler submodule and is built by EDG's own
# CMake project as a static library (edg_add_frontend_library, see
# edg_compiler/cmake/frontend-library.cmake), configured for ROSE by the macro configuration
# set that edg_build/ generates.  edgRose, the EDG IL -> SageIII translator, links that
# library; both end up in RoseSourceCxxFrontendEdg.

set(EDG_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/edg_compiler")
if(NOT EXISTS "${EDG_SOURCE_DIR}/src/basics.h")
  message(FATAL_ERROR
    "EDG source not found in ${EDG_SOURCE_DIR}.\n"
    "Run: git submodule update --init src/frontend/CxxFrontend/EDG/edg_compiler")
endif()

# EDG's "disc_union" macro selects whether the IL's discriminated unions are real unions or
# structs; structs are easier to inspect in a debugger.
if(USE_ROSE_EDG_DEBUGGING_SUPPORT)
  set(EDG_DISC_UNION "struct")
else()
  set(EDG_DISC_UNION "union")
endif()

# 1. ROSE's macro configuration set "rose" (generated into the build tree; sets
#    ROSE_EDG_MACRO_CONF_ROOT).
add_subdirectory(edg_build)

# 2. EDG's CMake project, front-end library support only: no executables, tools or runtime
#    library, ROSE's configuration set instead of one of EDG's.
set(EDG_MACRO_CONF "rose" CACHE STRING "EDG macro configuration set (ROSE's)" FORCE)
set(EDG_MACRO_CONF_ROOT "${ROSE_EDG_MACRO_CONF_ROOT}" CACHE PATH
    "Directory holding the EDG macro configuration sets (generated by ROSE)" FORCE)
set(EDG_BUILD_EXECUTABLES OFF CACHE BOOL "Build EDG's cpfe/cpfe-cp/cdisp (not in ROSE)" FORCE)
set(EDG_BUILD_TOOLS OFF CACHE BOOL "Build EDG's developer tools (not in ROSE)" FORCE)
set(EDG_BUILD_LIB OFF CACHE BOOL "Build EDG's runtime library (not in ROSE)" FORCE)
add_subdirectory(edg_compiler)

# 3. The front end as a library, configured for ROSE.  MAKE_FRONT_END_CALLABLE (edgRose calls
#    cfe_main) and disc_union change the IL's layout, so they are PUBLIC like the rest of the
#    configuration: edgRose must see exactly what the library was built with.
edg_add_frontend_library(edg_frontend rose)
target_compile_definitions(edg_frontend PUBLIC
  MAKE_FRONT_END_CALLABLE=TRUE
  disc_union=${EDG_DISC_UNION})
# The objects end up in the shared library RoseSourceCxxFrontendEdg.
set_target_properties(edg_frontend PROPERTIES POSITION_INDEPENDENT_CODE ON)

# 4. The translator.
add_subdirectory(edgRose)

# All of the library's objects go into the shared library (as the former edg_internal object
# library did), so EDG symbols stay available to everything that linked RoseSourceCxxFrontendEdg
# before; linking the library as well brings its usage requirements.
add_library(RoseSourceCxxFrontendEdg SHARED
    dummyCppFileForLibroseEDG.C
    $<TARGET_OBJECTS:edg_rose>
    $<TARGET_OBJECTS:edg_frontend>)
target_link_libraries(RoseSourceCxxFrontendEdg PRIVATE edg_frontend)

# edgRose calls into SageIII, which lives in librose, and librose in turn links this library. The Mach-O linker
# resolves every symbol of a shared library at link time (ELF does not), so on macOS let the unresolved SageIII
# symbols be looked up dynamically when librose loads this library.
if(APPLE)
  target_link_options(RoseSourceCxxFrontendEdg PRIVATE "LINKER:-undefined,dynamic_lookup")
endif()

set_target_properties(RoseSourceCxxFrontendEdg PROPERTIES
    INSTALL_RPATH "$ORIGIN;$ORIGIN/../lib"
    BUILD_WITH_INSTALL_RPATH FALSE
    INSTALL_RPATH_USE_LINK_PATH TRUE
    OUTPUT_NAME "RoseSourceCxxFrontendEdg")

install(TARGETS RoseSourceCxxFrontendEdg
    LIBRARY DESTINATION lib
    RUNTIME DESTINATION bin)
